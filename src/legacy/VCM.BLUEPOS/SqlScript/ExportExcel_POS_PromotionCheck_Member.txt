CREATE TABLE #TempPro (
						OfferNo nvarchar(20),

INSERT INTO #TempPro

SELECT  OfferNo,[No],N.BarcodeNo,UnitOfMeasure,DiscountType,DiscountValue
FROM
	(
		SELECT OfferNo,[No],UnitOfMeasure,DiscountType,DiscountValue
		FROM OfferBuy (Nolock)
		WHERE (CHARINDEX('[@ItemNo]',[No]) > 0 OR '[@ItemNo]' = '')
		UNION
		SELECT OfferNo,[No],UnitOfMeasure,DiscountType,DiscountValue
		FROM OfferGet (Nolock)
		WHERE (CHARINDEX('[@ItemNo]',[No]) > 0 OR '[@ItemNo]' = '')
		UNION
		SELECT OfferNo,[No],UnitOfMeasure,[ValueType] AS DiscountType,[Value] AS DiscountValue
		FROM OfferBenefits (Nolock)
		WHERE (CHARINDEX('[@ItemNo]',[No]) > 0 OR '[@ItemNo]' = '')
	) M	INNER JOIN Barcodes(Nolock) N on M.[No]= N.[ItemNo]

						H.[PromotionNo],

						H.[SalesType],

						M.[ItemNo],
						I.[Description] AS ItemName,
						M.[UOM],
						M.[DiscountType],
						CONVERT(float,M.[DiscountValue]) AS DiscountValue,
						-- Áp dụng cho thẻ thành viên
						H.[MemberOnly] AS MemberOnly,
						-- Hạng thẻ / Loại thẻ
						H.[MemberCode],  
						CONVERT(float,H.[LimitQty]) AS LimitQty,
						CONVERT(varchar(30),H.[VoucherFromDate],103) AS VoucherFromDate,
						CONVERT(varchar(30),H.[VoucherToDate],103) AS VoucherToDate,				
						CONVERT(varchar(30),H.[LastDateModified],103) AS LastDateModified	

		FROM [OfferHeader] H (NOLOCK) 
		LEFT JOIN [OfferSite] K (NOLOCK) ON H.[No] = K.[OfferNo] 
		LEFT JOIN #TempPro M (NOLOCK) ON H.[No]=M.[OfferNo]
		LEFT JOIN [dbo].[Item] I (NOLOCK) ON I.[No] = M.[ItemNo]
		LEFT JOIN [dbo].[SalesOrderType] S (NOLOCK) ON H.[SalesType] = S.[Code]
		WHERE ('[@StoreNo]' = '' OR '[@StoreNo]' = K.[StoreNo])	
		AND ('[@No]' = '' OR '[@No]' = H.[No] OR '[@No]' = H.[PromotionNo])
		AND ('[@OfferType]' = '' OR '[@OfferType]' = H.[OfferType])
		AND ('[@SalesType]' = '' OR '[@SalesType]' = H.[SalesType])
		AND H.[MemberCode] LIKE N'%[@MemberCode]%'
		AND ('[@ItemNo]' = '' OR '[@ItemNo]' = M.[ItemNo])
		AND ('[@KeyWord]' = '' OR Convert(varchar(50),'[@KeyWord]') = M.[Barcode] OR H.[Description] LIKE N'%[@KeyWord]%')
		
)T

		-- Giá = 0 ---> Thành tiền trên Web = 0
		CASE WHEN (T.SalesPrice <= 0 OR T.SalesPrice = '') THEN 0    -- Truong hop ngay hieu luc CTKM < Ngay hieu luc cua san pham trong bang gia
		     WHEN (T.Amount <= 0 OR T.Amount = '') THEN T.Amount

		END AS Amount
